Procedural UI Sounds.
Sleek. Tactile. Stark Tech.

No samples. No chiptune. No alien blips. Pure Web Audio synthesis — glassy transients, filtered noise, FM shimmer, tiny plate reverb. Built for premium interfaces.

Usage

import { frontierAudio } from '@frontier/ui-audio'

// Must be called after user gesture (click)
await frontierAudio.init({ masterVolume: 0.85, reverbAmount: 0.22 })

// Play any sound
frontierAudio.play('click')
frontierAudio.play('success', { pitch: 1.1, volume: 0.9 })
frontierAudio.play('hover', { velocity: 0.6 })

// Auto-bind via data attributes
// <button data-ui-sound="click" data-ui-sound-hover="hover">Click</button>
frontierAudio.bindAutoSounds()

// Custom procedural sound
frontierAudio.define('myBlip', {
  description: 'Custom',
  category: 'tactile',
  render: (ctx, engine, opts) => {
    engine.playTone({
      freq: 1200 * opts.pitch,
      waveform: 'sine-tri',
      attack: 0.004,
      decay: 0.18,
      gain: 0.5 * opts.volume,
      filterFreq: 6000
    })
  }
})
